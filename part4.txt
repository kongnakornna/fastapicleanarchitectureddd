            class FakeRedis:
                def __init__(self):
                    self._data: dict[str, str] = {}

                async def set(self, k, v, ex=None, nx=False):
                    if nx and k in self._data:
                        return None
                    self._data[k] = v
                    return True

                async def delete(self, *keys):
                    n = 0
                    for k in keys:
                        if k in self._data:
                            del self._data[k]
                            n += 1
                    return n

                async def ping(self): return True

                async def scan_iter(self, match=None, count=200):
                    pat = (match or "*").rstrip("*")
                    for k in list(self._data):
                        if k.startswith(pat):
                            yield k

            async def test_set_get():
                c = RedisCache(FakeRedis())
                await c.set("k1", {"a": 1})
                assert await c.get("k1") == {"a": 1}

            async def test_get_or_set_caches():
                c = RedisCache(FakeRedis())
                calls = {"n": 0}

                    calls["n"] += 1
                    return [1, 2, 3]

                v1 = await c.get_or_set("k", loader)
                v2 = await c.get_or_set("k", loader)
                assert v1 == v2 == [1, 2, 3]
                assert calls["n"] == 1

        for f in files_to_check:
            if (self.root / f).exists():
                ok(f"{f.split('/')[-1]} ✓")
            else:
                issues.append(f"missing: {f}")

        app_file = self.root / self.app_py
        if app_file.exists():
            content = app_file.read_text(encoding="utf-8")
            for marker in ("register_bigo_openapi(app)", "bigo_router", "bigo_ws_router", "bigo_admin_router"):
                if marker in content:
                    ok(f"app.py: {marker} ✓")
                else:
                    issues.append(f"app.py missing: {marker}")

        for ver in ("V001", "V002", "V003", "V004", "V005"):
            d = self.root / self.sql_dir
            matches = list(d.glob(f"{ver}__*{self.module}*.sql")) if d.exists() else []
            if matches:
                ok(f"SQL {ver} ✓")
            else:
                issues.append(f"SQL {ver} not found")

        alembic_file = self.root / self.alembic_dir / "bigo_001_add_bigo_tables.py"
        if alembic_file.exists():
            ok("alembic bigo_001 ✓")
        else:
            issues.append(f"alembic not found: {alembic_file}")

        postman = self.root / f"docs/postman/{self.module}.json"
        if postman.exists():
            try:
                _json_mod.loads(postman.read_text(encoding="utf-8"))
                ok("postman.json valid ✓")
            except Exception as e:

            ok("ALL CHECKS PASSED ✓")

    def _update_app_py(self) -> None:
        app_file = self.root / self.app_py
        if not app_file.exists():
            warn(f"{self.app_py} not found — skipping")
            return

        content = app_file.read_text(encoding="utf-8")
        original = content

        imports = [
            ("from app.modules.bigo.presentation.router import router as bigo_router", "bigo_router"),
            ("from app.modules.bigo.presentation.ws_router import ws_router as bigo_ws_router", "bigo_ws_router"),
            ("from app.modules.bigo.presentation.management_router import admin_router as bigo_admin_router", "bigo_admin_router"),
            ("from app.modules.bigo.presentation.swagger import register_bigo_openapi", "register_bigo_openapi"),
        ]

        for imp, marker in imports:
            if imp not in content and marker not in content:
                lines = content.split("\n")
                last = 0
                for i, line in enumerate(lines):
                    if line.startswith("from ") or line.startswith("import "):
                        last = i
                lines.insert(last + 1, imp)
                content = "\n".join(lines)
                ok(f"added: {imp}")

        if "app.include_router(bigo_router" not in content:
            anchor = re.search(r"(app\.include_router\([^\n]*\n)", content)
            if anchor:
                insert_at = anchor.end(1)
                includes = (
                    "app.include_router(bigo_router, prefix=\"/api/v1\")\n"
                    "app.include_router(bigo_ws_router)\n"
                    "app.include_router(bigo_admin_router, prefix=\"/api/v1\")\n"
                )
                content = content[:insert_at] + includes + content[insert_at:]
                ok("included bigo routers")

        if "register_bigo_openapi(app)" not in content:
            content = content.rstrip() + "\n\nregister_bigo_openapi(app)\n"
            ok("called register_bigo_openapi(app)")

        if content != original:
            bak = app_file.with_suffix(".py.bak")
            bak.write_bytes(app_file.read_bytes())
            app_file.write_text(content, encoding="utf-8", newline="\n")
            ok(f"{self.app_py} updated")
        else:
            skip(f"{self.app_py} unchanged")

    def _update_env_py(self) -> None:
        env_file = self.root / self.env_py
        if not env_file.exists():
            warn(f"{self.env_py} not found — skipping")
            return
        content = env_file.read_text(encoding="utf-8")
        original = content
        marker = "# --- module bigo ---"
        if marker in content:
            skip("bigo models block already present")
            return
        block = f'''{marker}
try:
    from app.modules.bigo.infrastructure.models import (  # noqa: F401
        CacheStatModel, KafkaConsumerModel, KafkaQueueModel,
        KafkaTopicModel, MemoryLeakModel, MemorySnapshotModel,
        MetricModel, PipelineReportModel, ProfileModel, WSSessionModel,
    )
except ImportError:
    pass


'''
        anchor = "config = context.config"
        idx = content.find(anchor)
        if idx == -1:
            warn("Cannot find anchor in env.py — skipping")
            return
        content = content[:idx] + block + content[idx:]

        if content != original:
            bak = env_file.with_suffix(".py.bak")
            bak.write_bytes(env_file.read_bytes())
            env_file.write_text(content, encoding="utf-8", newline="\n")
            ok(f"{self.env_py} updated")

def main() -> int:
    parser = argparse.ArgumentParser(add_help=False)
    parser.add_argument("action", nargs="?", default="help")
    parser.add_argument("module", nargs="?", default="bigo")
    parser.add_argument("layer", nargs="?", default="6")
    parser.add_argument("prefix", nargs="?", default="bigo")
    parser.add_argument("--force", action="store_true")
    parser.add_argument("--project-root", default=".")
    parser.add_argument("--help", action="store_true")

    args, _ = parser.parse_known_args()

    if args.help or args.action == "help":
        print(HELP)
        return 0

    root = Path(args.project_root).resolve()
    if not root.exists():
        err(f"Project root not found: {root}")
        return 1

    gen = BigOModuleGenerator(
        project_root=root, module=args.module,
        layer=args.layer, prefix=args.prefix, force=args.force,
    )

    print()
    info("═" * 60)
    info(f"  MODULE  : {gen.module}")
    info(f"  LAYER   : {gen.layer} ({gen.layer_name})")
    info(f"  SCHEMA  : {SCHEMA}")
    info(f"  PREFIX  : {gen.prefix}_")
    info(f"  ACTION  : {args.action}")
    info(f"  VERSION : {VERSION}")
    info("═" * 60)

    action_map = {
        "create": gen.create_module,
        "activate": gen.activate_module,
        "sql": gen.create_sql,
        "alembic": gen.create_migration,
        "swagger": gen.create_swagger,
        "postman": gen.create_postman,
        "test": gen.create_tests,
        "update": gen.update_app,
        "update-env": gen.update_env,
        "verify": gen.verify,
        "all": gen.run_all,
    }

    if args.action not in action_map:
        err(f"Unknown action: {args.action}")
        print(HELP)
        return 1

    try:
        action_map[args.action]()
    except Exception as e:
        err(f"Aborted: {e}")
        import traceback
        traceback.print_exc()
        return 1

    if args.action != "verify":
        gen.summary()
    return 0


if __name__ == "__main__":
